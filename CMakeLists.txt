cmake_minimum_required(VERSION 3.22)
project(liquidnat VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LIQUIDNAT_DPDK "Build actual DPDK packetd (no fallback)" ON)
option(LIQUIDNAT_SANITIZE "Address and undefined behaviour sanitizers" OFF)
option(LIQUIDNAT_FUZZ "Build libFuzzer parser target using Clang" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
add_compile_options("-ffile-prefix-map=${CMAKE_SOURCE_DIR}=/usr/src/liquidnat"
                    "-ffile-prefix-map=${CMAKE_BINARY_DIR}=/usr/src/liquidnat/build")
if(LIQUIDNAT_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(engine engine/src/packet.cpp engine/src/control.cpp)
target_include_directories(engine PUBLIC engine/include)
enable_testing()
add_executable(engine_tests tests/unit.cpp)
target_link_libraries(engine_tests PRIVATE engine)
add_test(NAME engine_unit COMMAND engine_tests)
set_tests_properties(engine_unit PROPERTIES TIMEOUT 30)
if(LIQUIDNAT_DPDK)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(DPDK REQUIRED IMPORTED_TARGET libdpdk)
  add_executable(packetd engine/src/main.cpp)
  target_link_libraries(packetd PRIVATE engine PkgConfig::DPDK)
endif()
if(LIQUIDNAT_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "Use clang++ for libFuzzer")
  endif()
  add_executable(parser_fuzz tests/parser_fuzz.cpp engine/src/packet.cpp engine/src/control.cpp)
  target_include_directories(parser_fuzz PRIVATE engine/include)
  target_compile_options(parser_fuzz PRIVATE -fsanitize=fuzzer,address,undefined -UNDEBUG)
  target_link_options(parser_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
